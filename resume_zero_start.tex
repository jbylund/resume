% Lato: narrower than DejaVu Sans, so it fits more words per line at a given size.
%
% Loaded by path from build/fonts/, NOT by system font name. Two reasons:
%
%  1. Correct text extraction. Lato's cmap aliases U+00AD and U+2010 onto the
%     same glyph as U+002D, and XeTeX's Identity-H embedding makes extraction
%     invert that mapping -- ambiguously, resolving to U+2010. The result is a
%     PDF with no ASCII hyphens in its text layer at all, so an ATS matches
%     neither the phone number nor `multi-tenant`. `make` runs
%     tools/unalias-hyphen.py to strip the two aliases into build/fonts/;
%     that script's docstring has the full account.
%  2. Reproducibility. \setmainfont{Lato} resolved by system name, so the text
%     layer silently depended on whichever Lato the build machine happened to
%     have. fonts/lato/ pins it; see fonts/README.md.
%
% Ligatures=TeX is what turns `---` into an em dash and `` '' into curly quotes.
% Stated explicitly rather than relying on the font: not every face carries the
% feature, and without it date ranges print a literal `---`.
\setmainfont{Lato-Regular.ttf}[
  Path            = build/fonts/,
  BoldFont        = Lato-Bold.ttf,
  ItalicFont      = Lato-Italic.ttf,
  BoldItalicFont  = Lato-BoldItalic.ttf,
  Numbers         = Lining,
  Ligatures       = TeX]

%%% Type scale %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%
% One knob for the whole document, as a percentage. Every size below is stated at
% its 100% value and scaled through \sz, so this is the only number to touch when
% the content grows or shrinks and the last page needs refilling.
%
% Raising this makes the text bigger AND the measure shorter (fewer characters per
% line), so pages fill faster than the percentage suggests. Measured against the
% current content and section order:
%
%    99%  ->  3 pages, last page 80% full
%   100%  ->  3 pages, last page 89% full
%   101%  ->  3 pages, last page 89% full   <-- current setting
%   102%  ->  spills onto a 4th page
%
% If content grows and the build fails the page check, drop a point rather than
% editing individual sizes.
%
% Was 107% before the header gained a links line and Open Source the typeshed
% entry, and 104% before Kensho gained the extraction, neighbour-search and
% entity-matching bullets. Each block of content costs two to three points.
%
% These figures are specific to the current section order. When Education sat in
% the middle of the document the safe range was not even contiguous -- 105, 106 and
% 109 all spilled while 107 and 108 fit -- because a section that cannot fit its
% \needlines reservation at the foot of a page pushes everything after it. Re-measure
% after moving sections around, not just after adding content.
\newcommand{\scalepct}{101}
\newcommand{\sz}[2]{%
  \fontsize{\the\dimexpr#1pt*\scalepct/100\relax}%
           {\the\dimexpr#2pt*\scalepct/100\relax}\selectfont}

\makeatletter
% \needlines{n} -- keep n lines' worth of space with what follows, else break
% before it, so a heading is never stranded at the foot of a page. This is
% needspace.sty's definition, inlined to avoid pulling in texlive-latex-extra
% for one macro.
\newcommand{\needlines}[1]{%
  \begingroup
    \setlength{\dimen@}{#1}%
    \vskip\z@\@plus\dimen@
    \penalty-100
    \vskip\z@\@plus-\dimen@
    \vskip\dimen@
    \penalty9999
    \vskip-\dimen@
    \vskip\z@skip
  \endgroup}
\makeatother

%%% Semantic markup %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% name, email, phone, location, profile links
%
% The links sit on their own line rather than extending the contact line: five
% bullet-separated fields overrun the measure at this size. Their link text is
% the bare domain, so the destination survives printing and text extraction --
% every other link in the document hides its URL behind a word.
\newcommand{\myheader}[5]{%
  \begin{center}
    {\sz{21}{24}\bfseries #1\par}
    \vspace{0.9ex}
    {\color{muted}\sz{9.5}{11} #2 \; \textbullet \; #3 \; \textbullet \; #4\par}
    \vspace{0.35ex}
    {\sz{9.5}{11} #5\par}
  \end{center}
  \vspace{0.4ex}
}

% Experience / Projects / Education / ... -- spaced capitals over a rule
\newcommand{\mysection}[1]{%
  \needlines{5\baselineskip}%
  \vspace{2.6ex}%
  {\sz{10.5}{12}\bfseries\color{accent}\MakeUppercase{#1}}\par
  \vspace{0.5ex}%
  {\color{accent}\hrule height \dimexpr0.7pt*\scalepct/100\relax}%
  \vspace{1.5ex}%
}

% organisation, location, role, dates
\newcommand{\myjob}[4]{%
  \needlines{4.5\baselineskip}%
  \vspace{1.2ex}%
  \begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}r@{}}
    {\sz{12.5}{14}\bfseries #1} & {\color{muted}\sz{10}{12} #2}\\[0.2ex]
    {\sz{10.5}{12}\itshape\color{muted} #3} & {\color{muted}\sz{10}{12} #4}\\
  \end{tabular*}
  \par\nopagebreak[4]\vspace{0.2ex}%
}

% a second qualification under the same institution
\newcommand{\mysubline}[1]{{\sz{10.5}{12}\itshape\color{muted} #1}\par}

% an open-source project heading
\newcommand{\mysubhead}[1]{%
  \needlines{3.5\baselineskip}\vspace{1.6ex}{\sz{11}{13}\bfseries #1}\par\vspace{0.3ex}}

% the "Thesis:" lead-in
\newcommand{\mylead}[1]{{\bfseries #1}}

% The topic a bullet is about, e.g. \blead{Query service} founding engineer on...
% Bold so a reader skimming gets the list of capabilities without reading the prose.
\newcommand{\blead}[1]{\textbf{#1} --- }

                   leftmargin=1.3em, labelsep=0.55em, labelwidth=0.6em,
                   label={\color{accent}\textbullet}]}
  {\end{itemize}}
